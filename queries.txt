-- Un año de derecho por empleado y aniversario. dias_derecho y fechas son snapshot.
CREATE TABLE [RH].[vacation_entitlements](
    [id_vacation_entitlement]     [int] IDENTITY(1,1) NOT NULL,
    [id_empleado]                 [int]           NOT NULL,
    [anio_servicio]               [int]           NOT NULL,   -- 1 = primer aniversario
    [fecha_aniversario]           [date]          NOT NULL,   -- DATEADD(YEAR, anio_servicio, fecha_ingreso)
    [fecha_limite_solicitud]      [date]          NOT NULL,   -- DATEADD(MONTH, 6, fecha_aniversario)
    [fecha_siguiente_aniversario] [date]          NOT NULL,   -- DATEADD(YEAR, anio_servicio + 1, fecha_ingreso)
    [dias_derecho]                [int]           NOT NULL,   -- snapshot del catálogo
    -- Bloque único (todo NULL o todo lleno)
    [fecha_inicio]                [date]          NULL,
    [fecha_fin]                   [date]          NULL,       -- último día de vacaciones, calculado con el horario
    [dias_tomados]                [int]           NULL,
    [solicitado_by]               [int]           NULL,
    [solicitado_at]               [datetime2](0)  NULL,       -- captura original; NO cambia al reprogramar
    -- Pago de días sobrantes (todo NULL o todo lleno)
    [fecha_pago]                  [date]          NULL,
    [dias_pagados]                [int]           NULL,
    [comentario_pago]             [nvarchar](500) NULL,
    [pagado_by]                   [int]           NULL,
    [pagado_at]                   [datetime2](0)  NULL,
    [created_at]                  [datetime2](0)  NOT NULL,   -- buildDate(new Date())
 CONSTRAINT [PK_vacation_entitlements] PRIMARY KEY CLUSTERED ([id_vacation_entitlement] ASC),
 CONSTRAINT [UQ_vacation_entitlements_empleado_anio] UNIQUE ([id_empleado], [anio_servicio]),
 CONSTRAINT [FK_vacation_entitlements_empleado] FOREIGN KEY ([id_empleado])
     REFERENCES [RH].[empleados] ([id_empleado]),
 CONSTRAINT [CK_vacation_entitlements_ventana] CHECK (
     [anio_servicio] >= 1 AND [dias_derecho] > 0
     AND [fecha_limite_solicitud] > [fecha_aniversario]
     AND [fecha_siguiente_aniversario] > [fecha_limite_solicitud]),
 -- IS NOT NULL explícito: un CHECK que da UNKNOWN pasa.
 CONSTRAINT [CK_vacation_entitlements_bloque] CHECK (
     ([fecha_inicio] IS NULL AND [fecha_fin] IS NULL AND [dias_tomados] IS NULL
          AND [solicitado_by] IS NULL AND [solicitado_at] IS NULL)
  OR ([fecha_inicio] IS NOT NULL AND [fecha_fin] IS NOT NULL AND [dias_tomados] IS NOT NULL
          AND [solicitado_by] IS NOT NULL AND [solicitado_at] IS NOT NULL
          AND [dias_tomados] BETWEEN 1 AND [dias_derecho]
          AND [fecha_inicio] >= [fecha_aniversario]
          AND [fecha_fin] >= [fecha_inicio]
          AND [fecha_fin] < [fecha_siguiente_aniversario])),
 CONSTRAINT [CK_vacation_entitlements_pago] CHECK (
     ([fecha_pago] IS NULL AND [dias_pagados] IS NULL AND [pagado_by] IS NULL AND [pagado_at] IS NULL)
  OR ([fecha_pago] IS NOT NULL AND [dias_pagados] IS NOT NULL AND [pagado_by] IS NOT NULL AND [pagado_at] IS NOT NULL
          AND [dias_pagados] = [dias_derecho] - ISNULL([dias_tomados], 0)
          AND [dias_pagados] > 0))
) ON [PRIMARY]
GO

-- Spec 63: escalones de acumulación por empresa y frecuencia. Se aplican en ciclo:
-- ciclo = MAX(retardos); descuento = floor(A / ciclo) × días del escalón más alto
--                                  + días del escalón más alto con retardos <= A mod ciclo.
CREATE TABLE [payroll].[lateness_tiers](
    [id_lateness_tier]  [int] IDENTITY(1,1) NOT NULL,
    [id_empresa]        [int]          NOT NULL,
    [id_payment_period] [smallint]     NOT NULL,
    [retardos]          [smallint]     NOT NULL,
    [dias_descuento]    [decimal](3,1) NOT NULL,
 CONSTRAINT [PK_lateness_tiers] PRIMARY KEY CLUSTERED ([id_lateness_tier] ASC),
 CONSTRAINT [UQ_lateness_tiers_empresa_frecuencia_retardos] UNIQUE ([id_empresa], [id_payment_period], [retardos]),
 CONSTRAINT [FK_lateness_tiers_settings] FOREIGN KEY ([id_empresa])
     REFERENCES [payroll].[lateness_settings] ([id_empresa]),
 CONSTRAINT [FK_lateness_tiers_payment_period] FOREIGN KEY ([id_payment_period])
     REFERENCES [RH].[payment_periods] ([id_payment_period]),
 CONSTRAINT [CK_lateness_tiers_valores] CHECK (
     [retardos] > 0 AND [dias_descuento] > 0
     AND [dias_descuento] * 2 = FLOOR([dias_descuento] * 2))
     -- "no decreciente" se valida en zod y en la action: un CHECK no ve otras filas
) ON [PRIMARY]
GO

-- Semillas: semanal (SAT 02) {1 → 0.5, 2 → 1}; quincenal (SAT 04) {2 → 0.5, 4 → 1}.

JOIN (VALUES ('02', 1, 0.5), ('02', 2, 1.0), ('04', 2, 0.5), ('04', 4, 1.0))

    [approved_by] [int]          NULL,
    [approved_at] [datetime2](0) NULL,
    [paid_by]     [int]          NULL,
    [paid_at]     [datetime2](0) NULL;
GO
ALTER TABLE [payroll].[periods] ADD
    CONSTRAINT [FK_periods_approved_by] FOREIGN KEY ([approved_by]) REFERENCES [dbo].[users] ([id_user]),
    CONSTRAINT [FK_periods_paid_by]     FOREIGN KEY ([paid_by])     REFERENCES [dbo].[users] ([id_user]),
    -- El sello y el estatus no pueden contradecirse: 1 y 2 sin sello, 3 aprobado, 4 aprobado y pagado.
    CONSTRAINT [CK_periods_sello_estatus] CHECK (
        ([status] <= 2 AND [approved_by] IS NULL AND [approved_at] IS NULL
                       AND [paid_by] IS NULL AND [paid_at] IS NULL)
     OR ([status] = 3  AND [approved_by] IS NOT NULL AND [approved_at] IS NOT NULL
                       AND [paid_by] IS NULL AND [paid_at] IS NULL)
     OR ([status] = 4  AND [approved_by] IS NOT NULL AND [approved_at] IS NOT NULL
                       AND [paid_by] IS NOT NULL AND [paid_at] IS NOT NULL));
GO

-- Spec 94: bitácora de cambios de estatus del periodo. Solo las tres transiciones de aprobación.
-- Se borra junto con el periodo (que solo se puede borrar en estatus 1, donde no tiene filas).
CREATE TABLE [payroll].[period_status_log](
    [id_period_status_log] [int] IDENTITY(1,1) NOT NULL,
    [id_period]            [int]           NOT NULL,
    [status_anterior]      [tinyint]       NOT NULL,
    [status_nuevo]         [tinyint]       NOT NULL,
    [comentario]           [nvarchar](200) NULL,     -- obligatorio al desaprobar (3 -> 2)
    [changed_by]           [int]           NOT NULL,
    [changed_at]           [datetime2](0)  NOT NULL,
 CONSTRAINT [PK_period_status_log] PRIMARY KEY CLUSTERED ([id_period_status_log] ASC),
 CONSTRAINT [FK_period_status_log_period] FOREIGN KEY ([id_period])
     REFERENCES [payroll].[periods] ([id_period]) ON DELETE CASCADE,
 CONSTRAINT [FK_period_status_log_user] FOREIGN KEY ([changed_by])
     REFERENCES [dbo].[users] ([id_user]),
 CONSTRAINT [CK_period_status_log_transicion] CHECK (
     ([status_anterior] = 2 AND [status_nuevo] = 3)     -- aprobar
  OR ([status_anterior] = 3 AND [status_nuevo] = 4)     -- marcar como pagada
  OR ([status_anterior] = 3 AND [status_nuevo] = 2)),   -- desaprobar
 CONSTRAINT [CK_period_status_log_comentario] CHECK (
     [status_nuevo] <> 2
  OR ([comentario] IS NOT NULL AND LEN(LTRIM(RTRIM([comentario]))) >= 5))
) ON [PRIMARY]
GO

CREATE NONCLUSTERED INDEX [IX_period_status_log_period]
    ON [payroll].[period_status_log] ([id_period], [changed_at] DESC)
GO

-- Spec 85: productos cuyo id_supplier no existe en proveedores de su empresa
SELECT p.[id_product], p.[name], p.[id_supplier]
  FROM [CentroPodologico].[inventory].[Products] p
 WHERE p.[id_supplier] IS NOT NULL
   AND NOT EXISTS (
         SELECT 1
           FROM [CentroPodologico].[inventory].[proveedores] s
          WHERE s.[id_proveedor] = p.[id_supplier]
            AND s.[id_empresa]   = p.[id_empresa]
       );

UPDATE p
   SET p.[id_supplier] = NULL
  FROM [CentroPodologico].[inventory].[Products] p
 WHERE p.[id_supplier] IS NOT NULL
   AND NOT EXISTS (
         SELECT 1
           FROM [CentroPodologico].[inventory].[proveedores] s
          WHERE s.[id_proveedor] = p.[id_supplier]
            AND s.[id_empresa]   = p.[id_empresa]
       );
